  StreamSubscription<DocumentSnapshot<Map<String, dynamic>>>? _followingSub;

  bool _isLocked = false;
  bool _isHiddenFromExplore = false;
  bool _isPausedWork = false;

  @override
  void initState() {
    super.initState();
    worker = widget.worker;

    final currentUid = FirebaseAuth.instance.currentUser?.uid;

  @override
  void dispose() {
    _workerDocSub?.cancel();
    _followingSub?.cancel();

  }

  // -------- Firestore Listeners --------

  Future<void> _setupAuthAndAttachFollowListener() async {
    final user = FirebaseAuth.instance.currentUser;
    _viewerUid = user?.uid;
    await _attachFollowingListener();
  }

  Future<void> _attachWorkerListener() async {
    if (_profileUid.isEmpty) return;

    _workerDocSub?.cancel();
    _workerDocSub = FirebaseFirestore.instance
        .collection('users')
        .doc(_profileUid)
        .snapshots()
        .listen((snap) {
      if (!snap.exists) return;
      final data = snap.data() ?? {};
      if (!mounted) return;

      setState(() {
        // counts এখানে user doc থেকে আর পড়ছি না (Option B)
        _isLocked = (data['isLocked'] ?? false) == true;
        _isHiddenFromExplore = (data['isHiddenFromExplore'] ?? false) == true;
        _isPausedWork = (data['isPausedWork'] ?? false) == true;

        // subscription পড়া (nested or flat)
        if (data['subscription'] is Map &&
            (data['subscription']['tier'] is String)) {
          _subscriptionPlan = data['subscription']['tier'] as String;
        } else if (data['subscriptionPlan'] is String) {
          _subscriptionPlan = data['subscriptionPlan'] as String;
        } else {
          _subscriptionPlan = 'free';
        }

        // ✅ প্রোফাইল ইনফো রিয়েলটাইম আপডেট (image/name/location/price/message)
        final newName = (data['name'] ?? worker.name).toString();
        final newImage = (data['image'] ?? worker.image).toString();
        final newLocation = (data['location'] ?? data['address'] ?? worker.location).toString();
        final newPrice = (data['price'] ?? data['priceLabel'] ?? worker.price).toString();
        final newMessage = (data['about'] ?? worker.message).toString();

        worker = Worker(
          id: worker.id,
          name: newName.isNotEmpty ? newName : worker.name,
          role: worker.role,
          image: newImage.isNotEmpty ? newImage : worker.image,
          message: newMessage,
          medals: worker.medals,
          isVerified: worker.isVerified,
          rating: worker.rating,
          location: newLocation.isNotEmpty ? newLocation : worker.location,
          price: newPrice.isNotEmpty ? newPrice : worker.price,
        );
      });
    });
  }

  // Option B: ফলো কাউন্ট aggregate count() দিয়ে লোড
  Future<void> _loadFollowCounts() async {
    try {
      // প্রোফাইল ইউজারের followers
      final followersAgg = await FirebaseFirestore.instance
          .collection('users')
          .doc(_profileUid)
          .collection('followers')
          .count()
          .get();

      // count nullable হতে পারে, তাই ?? 0
      final int followers = followersAgg.count ?? 0;

      int following = 0;
      if (_viewerUid != null) {
        final followingAgg = await FirebaseFirestore.instance
            .collection('users')
            .doc(_viewerUid)
            .collection('following_workers')
            .count()
            .get();
        following = followingAgg.count ?? 0;
      }

      if (!mounted) return;
      setState(() {
        _followersCount = followers;
        _followingCount = following;
      });
    } catch (e) {
      // চাইলে লগ রাখুন
      // debugPrint('loadFollowCounts error: $e');
    }
  }

  // এটি আপনার স্ক্রিন ক্লাসের ভেতরে থাকবে
  Future<void> _handleChatOpen(BuildContext context, {
    required String otherUserId,
    required String name,
    required String image,
    required String role
  }) async {

    // ১. স্ক্রিনের ওপর লোডিং দেখানো
    showDialog(
      context: context,
      barrierDismissible: false,
      builder: (context) => const Center(child: CircularProgressIndicator(color: Colors.blue)),
    );

    try {
      // ২. আইডি নিয়ে আসা
      final convId = await FirestoreChatService.getOrCreateConversation(otherUserId: otherUserId);

      // ৩. লোডিং বন্ধ করা
      if (!mounted) return;
      Navigator.pop(context);

      // ৪. চ্যাট স্ক্রিনে পাঠানো
      Navigator.push(
        context,
        MaterialPageRoute(builder: (_) => ChatScreen(
          conversationId: convId,
          userName: name,
          userImage: image,
          userRole: role,
        )),
      );
    } catch (e) {
      if (!mounted) return;
      Navigator.pop(context); // লোডিং বন্ধ
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text("Error: $e")));
    }
  }

  Future<void> _attachFollowingListener() async {
    if (_viewerUid == null || widget.isOwner || worker.id.isEmpty) return;

    _followingSub?.cancel();
    final ref = FirebaseFirestore.instance
        .collection('users')
        .doc(_viewerUid)
        .collection('following_workers')
        .doc(worker.id);

    _followingSub = ref.snapshots().listen((snap) {
      if (!mounted) return;
      setState(() => _isFollowing = snap.exists);
    });
  }

  // Follow / Unfollow (Option B – সাবকলেকশন + count)
  Future<void> _toggleFollow() async {
    if (_viewerUid == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text("লগইন প্রয়োজন।")),
      );
      return;
    }
    if (worker.id.isEmpty) return;

    final db = FirebaseFirestore.instance;
    final viewerRef = db.collection('users').doc(_viewerUid);
    final workerRef = db.collection('users').doc(worker.id);

    final followingRef = viewerRef.collection('following_workers').doc(worker.id);
    final followerRef = workerRef.collection('followers').doc(_viewerUid);

    final next = !_isFollowing;

    // Optimistic UI
    setState(() {
      _isFollowing = next;
      _followersCount = next ? _followersCount + 1 : (_followersCount > 0 ? _followersCount - 1 : 0);
    });

    try {
      final batch = db.batch();
      if (next) {
        batch.set(followingRef, {
          'workerId': worker.id,
          'followedAt': FieldValue.serverTimestamp(),
        });
        batch.set(followerRef, {
          'viewerId': _viewerUid,
          'followedAt': FieldValue.serverTimestamp(),
        });
      } else {
        batch.delete(followingRef);
        batch.delete(followerRef);
      }
      await batch.commit();

      // সার্ভার ট্রুথ রিফ্রেশ
      await _loadFollowCounts();
    } catch (e) {
      if (!mounted) return;
      // revert
      setState(() {
        _isFollowing = !next;
        _followersCount = next ? (_followersCount > 0 ? _followersCount - 1 : 0) : _followersCount + 1;
      });
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text("অ্যাকশন ব্যর্থ: $e")),
      );
    }
  }

  String get _workerKey {
    // ✅ Owner হলে key হিসেবে _profileUid বেশি সঠিক (worker.id সবসময় uid নাও হতে পারে)
    if (_profileUid.isNotEmpty) return _profileUid;
    if (worker.id.isNotEmpty) return worker.id;
    if ((widget.phoneNumber ?? '').trim().isNotEmpty) {
      return widget.phoneNumber!.trim();
    }
    return worker.name;
  }

  String get _profileColorKey => _workerKey;

  Future<void> _loadProfileColor() async {
    final idx = await ProfileColorService.getColorIndex(_profileColorKey);
    if (!mounted) return;
    setState(() => _profileColorIndex = idx);
  }

  Future<void> _loadViewerRole() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      final role = prefs.getString('user_role'); // 'finder' না কি 'maker'
      if (!mounted) return;
      setState(() {
        // যদি বর্তমান ইউজার 'finder' হয়, তবে সে একজন ওয়ার্কার
        _viewerIsFinder = (role == 'finder');
      });
    } catch (_) {}
  }

  Future<void> _loadDetailsFromPrefs() async {
    try {
      final prefs = await SharedPreferences.getInstance();
      if (!mounted) return;
      setState(() {
        final exp = (prefs.getString('worker_experience') ?? '').trim();
        final lang = (prefs.getString('worker_languages') ?? '').trim();
        final avail = (prefs.getString('worker_availability') ?? '').trim();
        if (exp.isNotEmpty) _experience = exp;
        if (lang.isNotEmpty) _languages = lang;
        if (avail.isNotEmpty) _availability = avail;
      });
    } catch (_) {}
  }

  Future<void> _loadStatsAndReviews() async {
    try {
      final key = _workerKey;
      final completed = await CompletedWorkService.getCompletedCountForWorker(key);
      final summary = await RatingHistoryService.getSummaryForWorker(key);
      final recent = await RatingHistoryService.getRecentReviewsForWorker(key, limit: 3);
      if (!mounted) return;
      setState(() {
        _completedCount = completed;
        _avgRating = (summary['avgRating'] as num?)?.toDouble() ?? worker.rating;
        _totalReviews = (summary['totalReviews'] as int?) ?? 0;
        _recentReviews = recent;
      });
    } catch (_) {}
  }

  bool get _isTopRated => (_avgRating ?? worker.rating) >= 4.8;
  bool get _isTrusted => (_avgRating ?? worker.rating) >= 4.2;
  bool get _hasBronze => (_avgRating ?? worker.rating) >= 3.5;
  bool get _hasSilver => (_avgRating ?? worker.rating) >= 4.0;
  bool get _hasGold => (_avgRating ?? worker.rating) >= 4.5;
  bool get _hasPlatinum => (_avgRating ?? worker.rating) >= 4.7;
  bool get _hasDiamond => (_avgRating ?? worker.rating) >= 4.9;

  Future<void> _launchUri(
      BuildContext context,
      Uri uri, {
        String errorMessage = 'Could not open app.',
      }) async {
    try {
      if (await canLaunchUrl(uri)) {
        await launchUrl(uri, mode: LaunchMode.externalApplication);
      } else {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text(errorMessage), backgroundColor: Colors.red),
        );
      }
    } catch (_) {}
  }

  Future<void> _callNow(BuildContext context) async {
    final phone = widget.phoneNumber?.trim();
    if (phone == null || phone.isEmpty) return;
    final uri = Uri(scheme: 'tel', path: phone);
    await _launchUri(context, uri, errorMessage: 'Could not open dialer.');
  }

  Future<void> _openUrl(BuildContext context, String? url) async {
    if (url == null || url.trim().isEmpty) return;
    final u = Uri.tryParse(url.trim());
    if (u == null) return;
    if (u.scheme == 'http' || u.scheme == 'https') {
      await _launchUri(context, u);
    }
  }

  Future<void> _sendEmail(BuildContext context) async {
    final email = widget.emailAddress?.trim();
    if (email == null || email.isEmpty) return;
    final uri = Uri(scheme: 'mailto', path: email);
    await _launchUri(context, uri, errorMessage: 'Could not open email app.');
  }

  ImageProvider? _getAvatarProvider() {
    final url = worker.image.trim();
    if (url.isNotEmpty &&
        (url.startsWith('http://') || url.startsWith('https://'))) {
      return NetworkImage(url);
    }
    return null;
  }

      backgroundColor: Colors.transparent,

        return Container(
          margin: const EdgeInsets.all(16),
          padding: const EdgeInsets.all(16),
          decoration: BoxDecoration(
            color: Colors.white,
            borderRadius: BorderRadius.circular(16),
          ),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Text(
                'Choose profile card theme',
                style: TextStyle(
                  fontWeight: FontWeight.bold,
                  color: AppColors.brandDark,
                ),
              ),
              const SizedBox(height: 12),

                      margin: const EdgeInsets.symmetric(horizontal: 6),

                      ),

              const SizedBox(height: 10),
            ],
          ),
        );
      },
    );

    if (selected != null) {
      await ProfileColorService.setColorIndex(_profileColorKey, selected);
      if (!mounted) return;
      setState(() => _profileColorIndex = selected);
    }
  }

  Future<void> _onOwnerMenuSelected(BuildContext context, _WorkerMenuOwner action) async {
    if (action == _WorkerMenuOwner.viewPublicCard) {
      _showPublicCardPreviewSheet();
      return;
    }

    if (_isPlanFree) {
      _showUpgradeDialog();
      return;
    }

    final uid = FirebaseAuth.instance.currentUser?.uid;
    if (uid == null) return;
    final userDoc = FirebaseFirestore.instance.collection('users').doc(uid);

    switch (action) {
      case _WorkerMenuOwner.lockAccount:
        try {
          final newValue = !_isLocked;
          await userDoc.update({'isLocked': newValue});
          if (!mounted) return;
          setState(() => _isLocked = newValue);
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text(newValue ? 'প্রোফাইল লক করা হয়েছে।' : 'প্রোফাইল আনলক করা হয়েছে।')),
          );
        } catch (e) {
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text('ফেইল্ড: $e')),
          );
        }
        break;

      case _WorkerMenuOwner.theme:
        _showThemeBottomSheet(context);
        break;

      case _WorkerMenuOwner.hideProfile:
        try {
          final newValue = !_isHiddenFromExplore;
          await userDoc.update({'isHiddenFromExplore': newValue});
          if (!mounted) return;
          setState(() => _isHiddenFromExplore = newValue);
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text(newValue ? 'প্রোফাইল Explore থেকে হাইড।' : 'প্রোফাইল Explore-এ ভিজিবল।')),
          );
        } catch (e) {
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text('ফেইল্ড: $e')),
          );
        }
        break;

      case _WorkerMenuOwner.pauseWork:
        try {
          final newValue = !_isPausedWork;
          await userDoc.update({'isPausedWork': newValue});
          if (!mounted) return;
          setState(() => _isPausedWork = newValue);
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text(newValue ? 'কাজ পজ করা হয়েছে।' : 'কাজ রিজিউম করা হয়েছে।')),
          );
        } catch (e) {
          ScaffoldMessenger.of(context).showSnackBar(
            SnackBar(content: Text('ফেইল্ড: $e')),
          );
        }
        break;

      default:
        break;
    }
  }

  Future<void> _onOtherMenuSelected(BuildContext context, _WorkerMenuOther action) async {
    switch (action) {
      case _WorkerMenuOther.report:
        Navigator.push(context, MaterialPageRoute(builder: (_) => const ReportScreen()));
        break;
      case _WorkerMenuOther.block:
        final idKey = _workerKey;
        await BlockedUserService().blockUser(idKey, worker.name);
        if (!mounted) return;
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('${worker.name} ব্লক করা হয়েছে।')),
        );
        break;
    }
  }

                    Text("Click to view or download", style: TextStyle(fontSize: 11, color: Colors.grey)),
                  ],
                ),
                Spacer(),
                Icon(Icons.visibility_outlined, color: Colors.grey),
              ],
            ),
          ),
        ),
        const SizedBox(height: 15),
        const Text("Certificates & ID", style: TextStyle(fontSize: 12, fontWeight: FontWeight.bold, color: Colors.grey)),
        const SizedBox(height: 8),
        SizedBox(
          height: 80,
          child: ListView.builder(
            scrollDirection: Axis.horizontal,
            itemCount: documents.length,
            itemBuilder: (context, index) {
              final url = documents[index];
              return GestureDetector(
                onTap: () => Navigator.push(context, MaterialPageRoute(builder: (_) => FullScreenImageView(imageUrl: url))),
                child: Container(
                  width: 100,
                  margin: const EdgeInsets.only(right: 10),
                  decoration: BoxDecoration(
                    borderRadius: BorderRadius.circular(8),
                    border: Border.all(color: Colors.grey.shade300),
                    image: DecorationImage(image: NetworkImage(url), fit: BoxFit.cover),
                  ),
                  child: Container(
                    decoration: BoxDecoration(
                      color: Colors.black.withOpacity(0.05),
                      borderRadius: BorderRadius.circular(8),
                    ),
                    child: const Center(child: Icon(Icons.zoom_in, color: Colors.white70)),
                  ),
                ),
              );
            },
          ),
        )
      ],
    );
  }

  Widget _buildSkillChip(String label) {
    return Chip(
      label: Text(label, style: const TextStyle(fontSize: 12, color: AppColors.brandDark)),
      backgroundColor: AppColors.brandLight,
      side: BorderSide.none,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(8)),
    );
  }

  Widget _buildPortfolioImage(BuildContext context, String url) {
    return GestureDetector(
      onTap: () => Navigator.push(context, MaterialPageRoute(builder: (_) => FullScreenImageView(imageUrl: url))),
      child: Container(
        margin: const EdgeInsets.only(right: 10),
        width: 120,
        decoration: BoxDecoration(
          borderRadius: BorderRadius.circular(10),
          image: DecorationImage(image: NetworkImage(url), fit: BoxFit.cover),
        ),
      ),
    );
  }

  Widget _buildInfoRow(IconData icon, String label, String value) {
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 8),
      child: Row(
        children: [
          Icon(icon, size: 20, color: Colors.grey),
          const SizedBox(width: 15),
          Text(label, style: const TextStyle(color: Colors.grey)),
          const Spacer(),
          Text(value, style: const TextStyle(fontWeight: FontWeight.w600, color: Colors.black87)),
        ],
      ),
    );
  }

  Widget _buildReviewItem(String name, String comment, int stars) {
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 8),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Row(
            children: [
              const Icon(Icons.account_circle, color: Colors.grey),
              const SizedBox(width: 10),
              Text(name, style: const TextStyle(fontWeight: FontWeight.bold)),
              const Spacer(),
              Row(
                children: List.generate(
                  5,
                      (i) => Icon(
                    Icons.star,
                    size: 14,
                    color: i < stars ? Colors.amber : Colors.grey[300],
                  ),
                ),
              )
            ],
          ),
          const SizedBox(height: 5),
          Text(comment, style: const TextStyle(fontSize: 13, color: Colors.black54)),
        ],
      ),
    );
  }
}

class FullScreenImageView extends StatelessWidget {
  final String imageUrl;
  const FullScreenImageView({super.key, required this.imageUrl});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.black,
      appBar: AppBar(
        backgroundColor: Colors.black,
        iconTheme: const IconThemeData(color: Colors.white),
        elevation: 0,
      ),
      body: Center(
        child: InteractiveViewer(
          child: Image.network(imageUrl, fit: BoxFit.contain),
        ),
      ),
    );
  }
}
